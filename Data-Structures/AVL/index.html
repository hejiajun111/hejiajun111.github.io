AVL树

AVL树是一种特殊的二叉搜索树,重点在于「平衡」.
先回顾二叉搜索树的定义:

  • 空树是二叉搜索树．
  • 若二叉搜索树的左子树不为空，则其左子树上所有点的附加权值均小于其根节点的值．
  • 若二叉搜索树的右子树不为空，则其右子树上所有点的附加权值均大于其根节点的值．
  • 二叉搜索树的左右子树均为二叉搜索树．

在AVL树中对于「平衡」的定义为: 如果 T 是一棵 AVL 树，当且仅当左右子树也是 AVL 树，且

即AVL树上任意一个节点的左右子树高度之差不超过

旋转

408常见伪代码

//LL 型：右旋
Node* RightRotate(Node* T)
{
    Node* L = T->left;
    T->left = L->right;
    L->right = T;

    return L;  // 返回新的子树根
}
//RR 型：左旋
Node* LeftRotate(Node* T)
{
    Node* R = T->right;
    T->right = R->left;
    R->left = T;

    return R;  // 返回新的子树根
}
//LR 型：先左旋，再右旋
Node* LRRotate(Node* T)
{
    T->left = LeftRotate(T->left);
    return RightRotate(T);
}
//RL 型：先右旋，再左旋
Node* RLRotate(Node* T)
{
    T->right = RightRotate(T->right);
    return LeftRotate(T);
}
//AVL 平衡调整
Node* Balance(Node* T)
{
    int bf = height(T->left) - height(T->right);

    if (bf > 1) {
        if (height(T->left->left) >= height(T->left->right))
            return RightRotate(T);  // LL
        else
            return LRRotate(T);     // LR
    }

    if (bf < -1) {
        if (height(T->right->right) >= height(T->right->left))
            return LeftRotate(T);   // RR
        else
            return RLRotate(T);     // RL
    }

    return T;  // 无需旋转
}

真实代码(无结构体,c指针)

// 节点编号从 1 开始，0 表示空节点
vector<int> val(1), lc(1), rc(1), ht(1);
int root = 0;

int height(int x)
{
    return x == 0 ? 0 : ht[x];
} 
void push_up(int x) // 根据左右孩子的高度更新当前节点高度
{
    ht[x] = max(height(lc[x]), height(rc[x])) + 1;
} 
int balance_factor(int x) // 平衡因子 = 左子树高度 - 右子树高度
{
    return height(lc[x]) - height(rc[x]);
} 
int rotate_right(int x) // 右旋：适用于 LL 型失衡
{
    int y = lc[x];
    lc[x] = rc[y]; // y 原来的右子树，转移给 x 作为左子树
    rc[y] = x;  // x 下降，成为 y 的右孩子
    push_up(x); // 先更新下降的 x，再更新上升的 y
    push_up(y);
    return y; // 返回旋转后的新根节点
}
int rotate_left(int x) // 左旋：适用于 RR 型失衡
{
    int y = rc[x];
    rc[x] = lc[y]; // y 原来的左子树，转移给 x 作为右子树
    lc[y] = x; // x 下降，成为 y 的左孩子

    push_up(x);
    push_up(y);

    return y;
}
 
int rebalance(int x) // 检查并恢复以 x 为根的子树的平衡
{
    if (balance_factor(x) > 1) // 左边过高
    {
        if (balance_factor(lc[x]) < 0)
        { 
            lc[x] = rotate_left(lc[x]); // LR：先将左孩子左旋，转化为 LL
        }
        return rotate_right(x); // LL 或处理完 LR：右旋当前节点
    }
    if (balance_factor(x) < -1) // 右边过高
    { 
        if (balance_factor(rc[x]) > 0)
        {
            rc[x] = rotate_right(rc[x]);  // RL：先将右孩子右旋，转化为 RR
        } 
        return rotate_left(x); // RR 或处理完 RL：左旋当前节点
    }
    return x;
}
 
int new_node(int v) // 创建新节点，返回其编号
{
    val.push_back(v);
    lc.push_back(0);
    rc.push_back(0);
    ht.push_back(1);
    return (int)val.size() - 1;
}
 
int insert(int x, int v) // 插入关键字 v，返回插入后子树的新根
{
    if (x == 0)
        return new_node(v);
    
    if (v < val[x]) // 先按照普通 BST 的规则插入
        lc[x] = insert(lc[x], v);
    else if (v > val[x])
        rc[x] = insert(rc[x], v);
    else
        return x; // 本示例不插入重复值
    push_up(x); // 更新当前节点高度

    return rebalance(x); // 检查是否失衡，必要时进行旋转
}

void inorder(int x) // 中序遍历，验证 BST 的有序性
{
    if (x == 0)
        return;
    inorder(lc[x]);
    cout << val[x] << ' ';
    inorder(rc[x]);
}
