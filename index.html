前缀函数

这是一个在线算法，输入一个字符串，找到每个前缀子串的最长公共前后缀。
重点在失配时的状态转移：
失配回退： 当前最长公共前后缀无法继续匹配时，尝试其次长公共前后缀，不断缩短 j，直到匹配成功或退回 0。
j = pi[j - 1]：利用已计算的前缀函数，直接跳到当前匹配前缀的最长真 Border（相等的前后缀）长度，跳过不可能的候选长度。

字典树 (Trie)

字典树（Trie） 是一种利用字符串公共前缀共享存储空间的树形数据结构。每条边代表一个字符，从根节点到某个节点的路径表示一个前缀，通过终止标记区分完整字符串与普通前缀。
算法流程

  • 初始化：创建根节点，每个节点维护字符对应的子节点及终止标记。
  • 插入：从根节点开始，依次遍历字符串的每个字符；若对应子节点不存在，则创建新节点，否则沿已有节点继续向下。遍历结束后标记字符串结尾。
  • 查询：从根节点开始，依次沿对应字符的边向下查找。若中途不存在对应子节点，则查询失败；若遍历结束，则根据终止标记判断字符串是否存在。
  • 核心思想：公共前缀共享节点，逐字符沿树转移。

#include <iostream>
#include <vector>
#include <string>
#include <algorithm>
#include <cassert>

class BigInt {
public:
    // 构造
    BigInt() : BigInt(0) {}
    BigInt(int x) {
        if (x < 0) negative = true, x = -x;
        else negative = false;
        if (x == 0) digits.push_back(0);
        else while (x) { digits.push_back(x % 10); x /= 10; }
    }
    BigInt(const std::string &s) {
        int start = 0;
        if (s[0] == '-') { negative = true; start = 1; }
        else negative = false;
        for (int i = s.size() - 1; i >= start; --i) {
            if (!isdigit(s[i])) throw std::invalid_argument("not a number");
            digits.push_back(s[i] - '0');
        }
        remove_leading_zeros();
        if (digits.empty()) digits.push_back(0), negative = false;
    }
    BigInt(const char *s) : BigInt(std::string(s)) {}

    // 比较运算符（绝对值先比较长度再逐位）
    bool operator==(const BigInt &rhs) const {
        return negative == rhs.negative && digits == rhs.digits;
    }
    bool operator!=(const BigInt &rhs) const { return !(*this == rhs); }
    bool operator<(const BigInt &rhs) const {
        if (negative != rhs.negative) return negative;
        if (digits.size() != rhs.digits.size())
            return (digits.size() < rhs.digits.size()) ^ negative;
        for (int i = digits.size() - 1; i >= 0; --i)
            if (digits[i] != rhs.digits[i])
                return (digits[i] < rhs.digits[i]) ^ negative;
        return false;
    }
    bool operator>(const BigInt &rhs) const { return rhs < *this; }
    bool operator<=(const BigInt &rhs) const { return !(rhs < *this); }
    bool operator>=(const BigInt &rhs) const { return !(*this < rhs); }

    // 算术运算符
    BigInt operator-() const {
        BigInt res = *this;
        if (!is_zero()) res.negative = !negative;
        return res;
    }
    BigInt operator+(const BigInt &rhs) const {
        if (negative == rhs.negative) {
            BigInt res = add_abs(*this, rhs);
            res.negative = negative;
            return res;
        } else {
            if (abs_compare(*this, rhs) >= 0) {
                BigInt res = sub_abs(*this, rhs);
                res.negative = negative;
                return res;
            } else {
                BigInt res = sub_abs(rhs, *this);
                res.negative = rhs.negative;
                return res;
            }
        }
    }
    BigInt operator-(const BigInt &rhs) const {
        return *this + (-rhs);
    }
    BigInt operator*(const BigInt &rhs) const {
        BigInt res;
        res.digits.assign(digits.size() + rhs.digits.size(), 0);
        for (size_t i = 0; i < digits.size(); ++i) {
            int carry = 0;
            for (size_t j = 0; j < rhs.digits.size(); ++j) {
                carry += res.digits[i + j] + digits[i] * rhs.digits[j];
                res.digits[i + j] = carry % 10;
                carry /= 10;
            }
            if (carry) res.digits[i + rhs.digits.size()] += carry;
        }
        res.remove_leading_zeros();
        res.negative = negative != rhs.negative;
        if (res.is_zero()) res.negative = false;
        return res;
    }
    BigInt operator/(const BigInt &rhs) const {
        if (rhs.is_zero()) throw std::domain_error("division by zero");
        if (abs_compare(*this, rhs) < 0) return BigInt(0);
        BigInt res, remainder;
        divide_abs(*this, rhs, res, remainder);
        res.negative = negative != rhs.negative;
        if (res.is_zero()) res.negative = false;
        return res;
    }
    BigInt operator%(const BigInt &rhs) const {
        if (rhs.is_zero()) throw std::domain_error("modulo by zero");
        BigInt res, remainder;
        divide_abs(*this, rhs, res, remainder);
        remainder.negative = negative;
        if (remainder.is_zero()) remainder.negative = false;
        return remainder;
    }

    // 复合赋值
    BigInt& operator+=(const BigInt &rhs) { *this = *this + rhs; return *this; }
    BigInt& operator-=(const BigInt &rhs) { *this = *this - rhs; return *this; }
    BigInt& operator*=(const BigInt &rhs) { *this = *this * rhs; return *this; }
    BigInt& operator/=(const BigInt &rhs) { *this = *this / rhs; return *this; }
    BigInt& operator%=(const BigInt &rhs) { *this = *this % rhs; return *this; }

    // 输出流
    friend std::ostream& operator<<(std::ostream &os, const BigInt &a) {
        if (a.negative && !a.is_zero()) os << '-';
        for (int i = (int)a.digits.size() - 1; i >= 0; --i)
            os << a.digits[i];
        return os;
    }

    // 转换为字符串
    std::string to_string() const {
        std::string res;
        if (negative && !is_zero()) res += '-';
        for (int i = (int)digits.size() - 1; i >= 0; --i)
            res += char('0' + digits[i]);
        return res;
    }

private:
    std::vector<int> digits;  // 低位在前，每个元素0-9
    bool negative = false;

    // 去掉前导零
    void remove_leading_zeros() {
        while (digits.size() > 1 && digits.back() == 0)
            digits.pop_back();
        if (digits.empty()) digits.push_back(0);
    }

    bool is_zero() const {
        return digits.size() == 1 && digits[0] == 0;
    }

    // 绝对值加法
    static BigInt add_abs(const BigInt &a, const BigInt &b) {
        BigInt res;
        res.digits.clear();
        int carry = 0;
        size_t i = 0;
        while (i < a.digits.size() || i < b.digits.size() || carry) {
            int sum = carry;
            if (i < a.digits.size()) sum += a.digits[i];
            if (i < b.digits.size()) sum += b.digits[i];
            res.digits.push_back(sum % 10);
            carry = sum / 10;
            ++i;
        }
        return res;
    }

    // 绝对值减法（要求 a >= b 且 a,b 非负）
    static BigInt sub_abs(const BigInt &a, const BigInt &b) {
        BigInt res;
        res.digits.clear();
        int borrow = 0;
        for (size_t i = 0; i < a.digits.size(); ++i) {
            int diff = a.digits[i] - borrow;
            if (i < b.digits.size()) diff -= b.digits[i];
            if (diff < 0) { diff += 10; borrow = 1; }
            else borrow = 0;
            res.digits.push_back(diff);
        }
        res.remove_leading_zeros();
        return res;
    }

    // 绝对值比较：返回 1 若 |a| > |b|, -1 若 <, 0 若相等
    static int abs_compare(const BigInt &a, const BigInt &b) {
        if (a.digits.size() != b.digits.size())
            return a.digits.size() < b.digits.size() ? -1 : 1;
        for (int i = (int)a.digits.size() - 1; i >= 0; --i)
            if (a.digits[i] != b.digits[i])
                return a.digits[i] < b.digits[i] ? -1 : 1;
        return 0;
    }

    // 除法（绝对值），商和余数
    static void divide_abs(const BigInt &a, const BigInt &b, BigInt &quot, BigInt &rem) {
        quot.digits.clear();
        rem = BigInt(0);
        for (int i = (int)a.digits.size() - 1; i >= 0; --i) {
            rem.digits.insert(rem.digits.begin(), a.digits[i]);
            rem.remove_leading_zeros();
            // 试商
            int q = 0;
            int l = 0, r = 9;
            while (l <= r) {
                int mid = (l + r) / 2;
                BigInt t = b * BigInt(mid);
                if (abs_compare(rem, t) >= 0) {
                    q = mid;
                    l = mid + 1;
                } else {
                    r = mid - 1;
                }
            }
            quot.digits.insert(quot.digits.begin(), q);
            BigInt sub = b * BigInt(q);
            rem = sub_abs(rem, sub);
        }
        quot.remove_leading_zeros();
        rem.remove_leading_zeros();
    }
};

class DSU {
public:
    explicit DSU(int n) : parent(n), size(n, 1) {
        std::iota(parent.begin(), parent.end(), 0);
    }

    int find(int x) {
        if (parent[x] != x)
            parent[x] = find(parent[x]);
        return parent[x];
    }

    bool unite(int x, int y) {
        x = find(x), y = find(y);
        if (x == y) return false;
        if (size[x] < size[y]) std::swap(x, y);
        parent[y] = x;
        size[x] += size[y];
        return true;
    }
    bool same(int x, int y) {
        return find(x) == find(y);
    }

    int get_size(int x) {
        return size[find(x)];
    }
    // 获取所有根节点
    std::vector<int> roots() const {
        std::vector<int> res;
        for (int i = 0; i < (int)parent.size(); ++i)
            if (parent[i] == i) res.push_back(i);
        return res;
    }

private:
    std::vector<int> parent;
    std::vector<int> size;
};

// ---------- 带权并查集 (Weighted DSU) ----------
// 权值类型 Weight 一般用 int / long long，需支持加法、减法运算
template<typename Weight>
class WeightedDSU {
public:
    explicit WeightedDSU(int n) : parent(n), weight(n, Weight()), size(n, 1) {
        std::iota(parent.begin(), parent.end(), 0);
    }

    // 查找根节点，同时更新 weight[x] 为从 x 到根节点的累积权值
    // 约定：weight[x] 表示 x 到 parent[x] 的权值，经过路径压缩后变为到根节点的权值
    int find(int x) {
        if (parent[x] != x) {
            int root = find(parent[x]);
            // 更新 x 到新根的距离：原先 x -> parent[x] 的权值 + parent[x] -> root 的权值
            weight[x] = weight[x] + weight[parent[x]];
            parent[x] = root;
        }
        return parent[x];
    }

    bool unite(int x, int y, Weight delta) {
        int rx = find(x), ry = find(y);
        if (rx == ry) {
            return (weight[x] - weight[y]) == delta;
        }
        if (size[rx] < size[ry]) {
            parent[rx] = ry;
            weight[rx] = delta + weight[y] - weight[x];
            size[ry] += size[rx];
        } else {
            parent[ry] = rx;
            weight[ry] = weight[x] - weight[y] - delta;
            size[rx] += size[ry];
        }
        return true;
    }

    Weight diff(int x, int y) {
        assert(find(x) == find(y));
        return weight[x] - weight[y];
    }
    bool same(int x, int y) {
        return find(x) == find(y);
    }
    // 获取元素到其根节点的权值
    Weight get_weight(int x) {
        find(x);
        return weight[x];
    }

private:
    std::vector<int> parent;
    std::vector<Weight> weight;
    std::vector<int> size;
};

struct EulerSieve {
    vector<int> primes;           // 素数列表
    vector<bool> is_composite;    // 是否为合数
    vector<int> min_factor;       // 最小质因子（可选）
    vector<int> phi;              // 欧拉函数（可选）

    // 构造函数，筛出 [0, n] 内的素数
    EulerSieve(int n) : is_composite(n + 1, false), min_factor(n + 1, 0), phi(n + 1, 0) {
        phi[1] = 1;
        for (int i = 2; i <= n; ++i) {
            if (!is_composite[i]) {
                primes.push_back(i);
                min_factor[i] = i;
                phi[i] = i - 1;          // 素数的欧拉函数 = i-1
            }
            for (int p : primes) {
                if (i * p > n) break;
                is_composite[i * p] = true;
                min_factor[i * p] = p;
                if (i % p == 0) {
                    // p 是 i 的最小质因子
                    phi[i * p] = phi[i] * p;
                    break;
                } else {
                    phi[i * p] = phi[i] * (p - 1);
                }
            }
        }
    }
    // 判断 x 是否为素数（x 需在筛的范围内）
    bool is_prime(int x) const {
        return !is_composite[x];
    }
};

// 需要类型 T 支持加法与减法（可用于前缀和）
template<typename T>
class Fenwick {
public:
    explicit Fenwick(int n) : n(n), bit(n + 1) {}
    // 单点增加
    void add(int i, T delta) {
        for (; i <= n; i += i & -i)
            bit[i] += delta;
    }
    // 前缀和 [1..i]
    T sum(int i) const {
        T res = 0;
        for (; i > 0; i -= i & -i)
            res += bit[i];
        return res;
    }
    // 区间和 [l..r]
    T query(int l, int r) const {
        return sum(r) - sum(l - 1);
    }
private:
    int n;
    std::vector<T> bit;
};

#include <vector>
using Matrix = std::vector<std::vector<long long>>;
Matrix mat_mul(const Matrix& A, const Matrix& B, long long mod) {
    int n = A.size();
    Matrix C(n, std::vector<long long>(n, 0));
    for (int i = 0; i < n; ++i)
        for (int k = 0; k < n; ++k)
            if (A[i][k] != 0)
                for (int j = 0; j < n; ++j)
                    C[i][j] = (C[i][j] + A[i][k] * B[k][j]) % mod;
    return C;
}
Matrix mat_pow(Matrix A, long long power, long long mod) {
    int n = A.size();
    Matrix res(n, std::vector<long long>(n, 0));
    for (int i = 0; i < n; ++i) res[i][i] = 1;  // 单位矩阵
    
    while (power) {
        if (power & 1) res = mat_mul(res, A, mod);
        A = mat_mul(A, A, mod);
        power >>= 1;
    }
    return res;
}
